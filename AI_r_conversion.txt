import pandas as pd

# Define helper functions
def fix_name(x):
    # Full list of all elements in GEOROC
    dict = ['SiO2', 'TiO2', 'Al2O3', ...]  # Rest of the list
    clean_name = x.replace('(WT%)', '').replace('(PPM)', '')
    p = [i for i, v in enumerate(dict) if v == clean_name.upper()][0]
    return dict[p]

def CleanData(data, elements_to_clean):
    data.rename(columns=lambda x: mapply(fix_name, x), inplace=True)
    data.dropna(subset=elements_to_clean, axis=0, inplace=True)
